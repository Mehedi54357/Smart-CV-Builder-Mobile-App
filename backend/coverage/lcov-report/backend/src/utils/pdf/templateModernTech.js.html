const axios = require('axios');
 
const fetchImage = async (url) => {
  try {
    const res = await axios.get(url, { responseType: 'arraybuffer', timeout: 10000 });
    return res.data;
  } catch (e) { return null; }
};

          doc.moveTo(0, i).lineTo(W, i).stroke();
